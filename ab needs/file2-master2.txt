TABLE: farmers
----------------------------------------------------------------------------
column              | type                     | constraints
----------------------------------------------------------------------------
id                  | INTEGER                  | PRIMARY KEY, AUTO INCREMENT
name                | VARCHAR(120)             | NOT NULL
name_gu             | VARCHAR(120)             | -
head_of_family      | VARCHAR(60)              | -
mobile              | VARCHAR(10)              | NOT NULL, UNIQUE
email               | VARCHAR(60)              | UNIQUE
village_id          | INTEGER                  | NOT NULL, FOREIGN KEY -> villages.id
house               | TEXT                     | -
upi_number          | VARCHAR(10)              | UNIQUE
bank_account_number | VARCHAR(30)              | UNIQUE
bank_name           | VARCHAR(30)              | -
bank_branch         | VARCHAR(30)              | -
ifsc_code           | VARCHAR(15)              | -
public_figure       | TEXT                     | -
description         | TEXT                     | -
created_at          | TIMESTAMP WITH TIME ZONE | NOT NULL, DEFAULT now()

Table constraints / indexes:
  CHECK ck_mobile_10_digits: mobile ~ '^[0-9]{10}$'
  CHECK ck_upi_number_10_digits: upi_number IS NULL OR upi_number ~ '^[0-9]{10}$'
  INDEX ix_farmers_village_id (village_id)
