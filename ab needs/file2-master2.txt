TABLE: farmers
----------------------------------------------------------------------------
column                  | type                     | constraints
----------------------------------------------------------------------------
id                      | INTEGER                  | PRIMARY KEY, AUTO INCREMENT
farmer_name             | VARCHAR(120)             | NOT NULL
farmer_name_gu          | VARCHAR(120)             | -
head_of_family          | VARCHAR(60)              | -
mobile                  | VARCHAR(10)              | NOT NULL, UNIQUE
email                   | VARCHAR(60)              | UNIQUE
village_id              | INTEGER                  | NOT NULL, FOREIGN KEY -> villages.id
house                   | TEXT                     | -
upi_number              | VARCHAR(10)              | UNIQUE
bank_account_number     | VARCHAR(30)              | UNIQUE
bank_name               | VARCHAR(30)              | -
bank_branch             | VARCHAR(30)              | -
ifsc_code               | VARCHAR(15)              | -
private_note_owner_only | TEXT                     | -
description             | TEXT                     | -
created_at              | TIMESTAMP WITH TIME ZONE | NOT NULL, DEFAULT now()

Table constraints / indexes:
  CHECK ck_mobile_10_digits: mobile ~ '^[0-9]{10}$'
  CHECK ck_upi_number_10_digits: upi_number IS NULL OR upi_number ~ '^[0-9]{10}$'
  INDEX ix_farmers_village_id (village_id)

TABLE: labour_group_wages
----------------------------------------------------------------------
column          | type                     | constraints
----------------------------------------------------------------------
id              | INTEGER                  | PRIMARY KEY, AUTO INCREMENT
labour_group_id | INTEGER                  | NOT NULL, FOREIGN KEY -> labour_groups.id
grain_id        | INTEGER                  | NOT NULL, FOREIGN KEY -> grains.id
weighing_rate   | INTEGER                  | NOT NULL
loading_rate    | INTEGER                  | NOT NULL
pulling_rate    | INTEGER                  | NOT NULL
mixing_rate     | INTEGER                  | NOT NULL
other_rate      | INTEGER                  | -
description     | TEXT                     | -
created_at      | TIMESTAMP WITH TIME ZONE | NOT NULL, DEFAULT now()

Table constraints / indexes:
  CHECK ck_labour_group_wages_rate_nonneg: weighing_rate >= 0 AND loading_rate >= 0 AND pulling_rate >= 0 AND mixing_rate >= 0
  INDEX ix_labour_group_wages_grain_id (grain_id)
  UNIQUE uq_labour_group_wages_group_grain (labour_group_id, grain_id)
