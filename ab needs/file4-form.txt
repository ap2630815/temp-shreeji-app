TABLE: commencings
-------------------------------------------------------------------------------
column          | type                     | constraints
-------------------------------------------------------------------------------
id              | INTEGER                  | PRIMARY KEY, AUTO INCREMENT
business_year   | VARCHAR(7)               | NOT NULL
commencing_date | DATE                     | NOT NULL
village         | VARCHAR(100)             | NOT NULL
grain_id        | INTEGER                  | NOT NULL, FOREIGN KEY -> grains.id
rate_per_20kg   | INTEGER                  | NOT NULL
net_weight      | INTEGER                  | NOT NULL
pay_after_days  | INTEGER                  | NOT NULL
terms           | TEXT                     | -
description     | TEXT                     | -
created_by      | INTEGER                  | FOREIGN KEY -> users.id
created_at      | TIMESTAMP WITH TIME ZONE | NOT NULL, DEFAULT now()

Table constraints / indexes:
  CHECK ck_commencings_net_weight: net_weight IN (35, 50, 70, 100)
  CHECK ck_commencings_rate_positive: rate_per_20kg > 0
  CHECK ck_commencings_pay_after: pay_after_days >= 0
  INDEX ix_commencings_business_year (business_year)

TABLE: bilties
--------------------------------------------------------------------------------------------------
column                | type                     | constraints
--------------------------------------------------------------------------------------------------
id                    | INTEGER                  | PRIMARY KEY, AUTO INCREMENT
trip_id               | INTEGER                  | NOT NULL, UNIQUE, FOREIGN KEY -> truck_trips.id
business_year         | VARCHAR(7)               | NOT NULL
bilty_date            | DATE                     | NOT NULL
bill_number           | INTEGER                  | NOT NULL
broker_id             | INTEGER                  | NOT NULL, FOREIGN KEY -> brokers.id
grain_rate            | INTEGER                  | NOT NULL
rate_unit_kg          | INTEGER                  | NOT NULL, DEFAULT 20
rate_change           | INTEGER                  | NOT NULL, DEFAULT 0
brokerage_amount      | INTEGER                  | NOT NULL, DEFAULT 0
merchant_total_weight | INTEGER                  | NOT NULL
merchant_tare_weight  | INTEGER                  | NOT NULL
industry_total_weight | INTEGER                  | -
industry_tare_weight  | INTEGER                  | -
description           | TEXT                     | -
created_by            | INTEGER                  | FOREIGN KEY -> users.id
created_at            | TIMESTAMP WITH TIME ZONE | NOT NULL, DEFAULT now()

Table constraints / indexes:
  CHECK ck_bilties_rate_change: grain_rate - rate_change >= 0
  CHECK ck_bilties_merchant_weights: merchant_total_weight > merchant_tare_weight AND merchant_tare_weight >= 0
  CHECK ck_bilties_industry_weights: (industry_total_weight IS NULL AND industry_tare_weight IS NULL) OR (industry_total_weight > industry_tare_weight AND industry_tare_weight >= 0)
  CHECK ck_bilties_rate: grain_rate > 0 AND rate_unit_kg > 0
  CHECK ck_bilties_brokerage: brokerage_amount >= 0
  CHECK ck_bilties_bill_number: bill_number > 0
  UNIQUE (business_year, bill_number)
  INDEX ix_bilties_business_year (business_year)
